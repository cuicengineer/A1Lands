ALTER PROCEDURE [dbo].[sp_GetContractInvoiceSchedule]
    @ContractNo NVARCHAR(50) = NULL,
    @FromDate DATE = NULL,
    @ToDate DATE = NULL,
    @CmdId INT = NULL,
    @ClassId INT = NULL,
    @BaseId INT = NULL
AS
BEGIN
    SET NOCOUNT ON;

    SELECT *
    FROM dbo.fn_GetContractInvoiceSchedule() f


    WHERE

        /* =====================================================
           CONTRACT NO FILTER
        ===================================================== */
        (@ContractNo IS NULL OR f.ContractNo = @ContractNo)

        /* =====================================================
           CMD / CLASS / BASE FILTERS
           (assuming these exist in Contracts table)
        ===================================================== */
        AND (@CmdId IS NULL OR f.CmdId = @CmdId)
        AND (@ClassId IS NULL OR f.ClassId = @ClassId)
        AND (@BaseId IS NULL OR f.BaseId = @BaseId)

        /* =====================================================
           DATE FILTERS (INVOICE PERIOD BASED)
        ===================================================== */
        AND (@FromDate IS NULL OR f.PeriodEnd >= @FromDate)
        AND (@ToDate IS NULL OR f.PeriodStart <= @ToDate);

END
GO

ALTER FUNCTION [dbo].[fn_BasicContractInfo](@AsOfDate date)
RETURNS TABLE
AS
RETURN
(
    SELECT
        c.Id,
        c.ContractNo,
		 c.commercialoperationdate,
        ContractState =
        CASE
            WHEN @AsOfDate < c.ContractStartDate THEN 'Pre-Mature'
            WHEN @AsOfDate > c.ContractEndDate THEN 'Terminated'
            WHEN DATEDIFF(MONTH, @AsOfDate, c.ContractEndDate) < 4 THEN 'Expiring'
            ELSE 'Valid'
        END,

        RRFY = pg.Fiscal,

        calc2.AHQShare,
        calc2.RACShare,
        calc2.BaseShare,

        FY =
        CASE
            WHEN @AsOfDate > c.ContractEndDate
            THEN
                CASE
                    WHEN MONTH(c.ContractEndDate) > 6
                        THEN CAST(YEAR(c.ContractEndDate) AS VARCHAR(4))
                             + '-' + RIGHT(CAST(YEAR(c.ContractEndDate) + 1 AS VARCHAR(4)), 2)
                    ELSE CAST(YEAR(c.ContractEndDate) - 1 AS VARCHAR(4))
                         + '-' + RIGHT(CAST(YEAR(c.ContractEndDate) AS VARCHAR(4)), 2)
                END
            ELSE
                CASE
                    WHEN MONTH(@AsOfDate) > 6
                        THEN CAST(YEAR(@AsOfDate) AS VARCHAR(4))
                             + '-' + RIGHT(CAST(YEAR(@AsOfDate) + 1 AS VARCHAR(4)), 2)
                    ELSE CAST(YEAR(@AsOfDate) - 1 AS VARCHAR(4))
                         + '-' + RIGHT(CAST(YEAR(@AsOfDate) AS VARCHAR(4)), 2)
                END
        END,

        ISNULL(base.AreaBase, -99) AS AreaBase,
        ISNULL(base.RateBase, -99) AS GroupRate,

        CAST(ROUND(rvRate.Rate,0) AS BIGINT) AS RentalValueRate,
        rvRate.ApplicableDate AS Auto_RentalValueRateAppDate,

        CAST(ROUND(govRate.Rate,0) AS BIGINT) AS Auto_GovtSharePercent,
        govRate.ApplicableDate AS Auto_GovtShareRateAppDate,

        calc1.RentalValue,
        calc1.GovtShare,
        calc1.PAFShare,

        Viability =
        CASE
            WHEN c.InitialRentPA IS NOT NULL
                 AND calc1.GovtShare IS NOT NULL
                 AND c.InitialRentPA <= calc1.GovtShare
            THEN N'Unviable'
            WHEN c.InitialRentPA IS NOT NULL
                 AND calc1.GovtShare IS NOT NULL
            THEN N'Viable'
            ELSE NULL
        END,

        0 AS due,
        0 AS paid,
        0 AS rcvable,
        'Overpaid' AS invoices,
		TenantNo
   FROM dbo.Contracts c

    LEFT JOIN dbo.Commands cmd
        ON cmd.Id = c.CmdId
        AND (cmd.IsDeleted = 0 OR cmd.IsDeleted IS NULL)

    LEFT JOIN dbo.Bases b
        ON b.Id = c.BaseId
        AND (b.IsDeleted = 0 OR b.IsDeleted IS NULL)

    LEFT JOIN dbo.Classes cls
        ON cls.Id = c.ClassId
        AND (cls.IsDeleted = 0 OR cls.IsDeleted IS NULL)

    /* =========================================================
       Revenue Rate
       ========================================================= */
    OUTER APPLY
    (
        SELECT TOP (1) pg.*
        FROM dbo.vw_LatestRevenueRates pg
        WHERE pg.Id = c.GrpId
          AND TRY_CAST(LEFT(pg.Fiscal, 4) AS INT) <=
              CASE WHEN MONTH(@AsOfDate) >= 6 THEN YEAR(@AsOfDate)
                   ELSE YEAR(@AsOfDate) - 1 END
        ORDER BY TRY_CAST(LEFT(pg.Fiscal, 4) AS INT) DESC
    ) pg

    /* =========================================================
       Rental Value Rate
       ========================================================= */
    OUTER APPLY
    (
        SELECT TOP (1) rv.*
        FROM dbo.RentalValueGovtShareRates rv
        WHERE rv.CmdId = c.CmdId
          AND rv.BaseId = c.BaseId
          AND rv.ClassId = c.ClassId
          AND rv.Type = 1
          AND (rv.IsDeleted = 0 OR rv.IsDeleted IS NULL)
          AND (rv.Status = 1 OR rv.Status IS NULL)
          AND rv.ApplicableDate <= @AsOfDate
          AND (rv.DeactiveDate IS NULL OR rv.DeactiveDate >= @AsOfDate)
        ORDER BY rv.ApplicableDate DESC
    ) rvRate

    /* =========================================================
       Govt Share Rate
       ========================================================= */
    OUTER APPLY
    (
        SELECT TOP (1) gv.*
        FROM dbo.RentalValueGovtShareRates gv
        WHERE gv.CmdId = c.CmdId
          AND gv.BaseId = c.BaseId
          AND gv.ClassId = c.ClassId
          AND gv.Type = 2
          AND (gv.IsDeleted = 0 OR gv.IsDeleted IS NULL)
          AND (gv.Status = 1 OR gv.Status IS NULL)
          AND gv.ApplicableDate <= @AsOfDate
          AND (gv.DeactiveDate IS NULL OR gv.DeactiveDate >= @AsOfDate)
        ORDER BY gv.ApplicableDate DESC
    ) govRate

    /* =========================================================
       Sharing Formula (LATEST ACTIVE)
       ========================================================= */
    OUTER APPLY
    (
        SELECT TOP (1)
        sf.AHQRate,
        sf.RACRate,
        sf.BaseRate
    FROM dbo.SharingFormulas sf
    WHERE sf.CmdId = c.CmdId
      AND sf.BaseId = c.BaseId
      AND sf.ClassId = c.ClassId
      AND (sf.IsDeleted = 0 OR sf.IsDeleted IS NULL)
      AND (sf.Status = 1 OR sf.Status IS NULL)
    ORDER BY sf.ApplicableDate DESC
    ) sf

    /* =========================================================
       BASE CALCULATION (NO SELF REFERENCE)
       ========================================================= */
    CROSS APPLY
    (
        SELECT
            AreaBase = COALESCE(pg.Area, c.GroupArea),
            RateBase = pg.Rate
    ) base

    /* =========================================================
       CALC 1: RENT + GOV + PAF (CORE LOGIC)
       ========================================================= */
    CROSS APPLY
    (
        SELECT
            RentalValue =
                CASE
                    WHEN base.AreaBase IS NOT NULL
                     AND base.RateBase IS NOT NULL
                     AND rvRate.Rate IS NOT NULL
                    THEN CAST(ROUND(base.AreaBase * base.RateBase * ROUND(rvRate.Rate,0) / 100.0,0) AS BIGINT)
                END,

            GovtShare =
                CASE
                    WHEN base.AreaBase IS NOT NULL
                     AND base.RateBase IS NOT NULL
                     AND rvRate.Rate IS NOT NULL
                     AND govRate.Rate IS NOT NULL
                     AND ISNULL(govRate.Config,'') <> 'Annual Rent'
                    THEN CAST(ROUND(base.AreaBase * base.RateBase * ROUND(rvRate.Rate,0) * ROUND(govRate.Rate,0) / 10000.0,0) AS BIGINT)
                    ELSE
                        CASE
                            WHEN govRate.Rate IS NOT NULL
                             AND govRate.Config = 'Annual Rent'
                            THEN CAST(ROUND(c.InitialRentPA * ROUND(govRate.Rate,0) / 100.0,0) AS BIGINT)
                        END
                END,

            PAFShare =
                CASE
                    WHEN base.AreaBase IS NOT NULL
                     AND base.RateBase IS NOT NULL
                     AND rvRate.Rate IS NOT NULL
                     AND govRate.Rate IS NOT NULL
                    THEN CAST(
                        ROUND(base.AreaBase * base.RateBase * ROUND(rvRate.Rate,0) / 100.0,0)
                        -
                        ROUND(base.AreaBase * base.RateBase * ROUND(rvRate.Rate,0) * ROUND(govRate.Rate,0) / 10000.0,0)
                    AS BIGINT)
                END
    ) calc1

    /* =========================================================
       CALC 2: SHARE SPLIT (SAFE - NO SELF REFERENCE)
       ========================================================= */
	   CROSS APPLY
(
    SELECT
        AHQRaw = calc1.PAFShare * sf.AHQRate / 100.0,
        RACRaw = calc1.PAFShare * sf.RACRate / 100.0,
        BaseRaw = calc1.PAFShare * sf.BaseRate / 100.0
) r
   CROSS APPLY
(
    SELECT
        AHQShare = CAST(FLOOR(r.AHQRaw) AS BIGINT),
        RACShare = CAST(FLOOR(r.RACRaw) AS BIGINT),
        BaseShare =
            CAST(
                calc1.PAFShare
                - FLOOR(r.AHQRaw)
                - FLOOR(r.RACRaw)
            AS BIGINT)
) calc2

    WHERE (c.IsDeleted = 0 OR c.IsDeleted IS NULL)
);
GO

ALTER FUNCTION [dbo].[fn_GetContractInvoiceSchedule]
(
)
RETURNS TABLE
AS
RETURN
(
    /* =========================================================
       1. BASE CONTRACTS
    ========================================================= */
    WITH Contracts AS
    (
        SELECT *
        FROM dbo.Contracts c
        WHERE ISNULL(c.IsDeleted,0) = 0 --and c.ApprovalStatus = 1
    ),

    /* =========================================================
       2. TALLY (controls max invoice cycles)
    ========================================================= */
    Tally AS
    (
        SELECT TOP (200)
            ROW_NUMBER() OVER (ORDER BY (SELECT NULL)) - 1 AS n
        FROM sys.all_objects
    ),

    /* =========================================================
       3. GENERATE BILLING PERIODS
       (PURE CONTRACT DRIVEN - NO INPUT PARAMETER)
    ========================================================= */
    BillingPeriods AS
    (
        SELECT
            c.Id AS ContractId, c.CmdId, c.BaseId, c.ClassId,
            c.ContractNo,  c.ContractStartDate, c.ContractEndDate, c.BusinessName, c.InitialRentPM,
			c.PaymentTermMonths, c.RiseTermType, cast(c.IncreaseIntervalMonths as varchar(10)) + ' Months' as RiseTerm, c.IncreaseRatePercent as RiseRate,
			c.RiseDate, 'Start of Period' as InvoiceDateType,
           

            DATEADD(MONTH, t.n * c.PaymentTermMonths, c.ContractStartDate) AS PeriodStart,
            DATEADD(MONTH, (t.n + 1) * c.PaymentTermMonths, c.ContractStartDate) AS PeriodEnd,

            t.n + 1 AS InvoiceSeq
        FROM Contracts c
        JOIN Tally t
            ON DATEADD(MONTH, t.n * c.PaymentTermMonths, c.ContractStartDate)
               <= c.ContractEndDate
			   where  c.IsDeleted = 0
    ),

    /* =========================================================
       4. CLAMP TO CONTRACT END DATE
    ========================================================= */
    Clamped AS
    (
        SELECT
            b.ContractId,b.ContractNo,
            b.InvoiceSeq,
			 b.ContractStartDate,b.ContractEndDate, b.BusinessName, b.InitialRentPM,
			b.PaymentTermMonths, b.RiseTermType, b.RiseTerm, b.RiseRate,
			b.RiseDate,  b.InvoiceDateType,b.CmdId, b.BaseId, b.ClassId
            ,
            CASE
                WHEN b.PeriodStart < c.ContractStartDate THEN c.ContractStartDate
                ELSE b.PeriodStart
            END AS PeriodStart,

            CASE
                WHEN b.PeriodEnd > c.ContractEndDate THEN c.ContractEndDate
                ELSE b.PeriodEnd
            END AS PeriodEnd
        FROM BillingPeriods b
        JOIN Contracts c ON c.Id = b.ContractId
        WHERE b.PeriodStart <= c.ContractEndDate 
		and c.IsDeleted = 0
    ),

    /* =========================================================
       5. APPLY RENT ENGINE (YOUR FUNCTION)
       ========================================================= */
    RentApplied AS
    (
        SELECT
            c.*,
            r.CalculatedRentPM,
            r.CalculatedRentPA
        FROM Clamped c
        CROSS APPLY dbo.fn_GetRunningContractsRentAsOfDate(c.PeriodStart) r
        WHERE r.ContractId = c.ContractId
    ),

    /* =========================================================
       6. DETECT RENT CHANGE BETWEEN PERIODS
    ========================================================= */
    Ordered AS
    (
        SELECT
            *,
            LAG(CalculatedRentPM) OVER
            (
                PARTITION BY ContractId
                ORDER BY PeriodStart
            ) AS PrevRentPM
        FROM RentApplied
    ),

    /* =========================================================
       7. MARK SEGMENTS
    ========================================================= */
    Segments AS
    (
        SELECT
            *,
            CASE
                WHEN PrevRentPM IS NULL THEN 1
                WHEN PrevRentPM <> CalculatedRentPM THEN 1
                ELSE 0
            END AS IsNewSegment
        FROM Ordered
    ),

    /* =========================================================
       8. FINAL OUTPUT (INVOICE READY)
    ========================================================= */
   Final AS
(
    SELECT  
        ContractId,
        ContractNo,

        CONCAT(FORMAT(InvoiceSeq,'000'), '-', ContractNo) AS InvoiceNo,

        PeriodStart,
        PeriodEnd,
		DATEFROMPARTS(
    YEAR(DATEADD(MONTH, 1, PeriodEnd)),
    MONTH(DATEADD(MONTH, 1, PeriodEnd)),
    10
) DueDate,

        DATEDIFF(MONTH, PeriodStart, PeriodEnd) AS Months,

        cast(CalculatedRentPM as int) CalculatedRentPM,

        cast( DATEDIFF(MONTH, PeriodStart, PeriodEnd) * CalculatedRentPM as int) AS TotalRent,

        /* =====================================================
           NEW: RENTER CHANGE REMARKS
        ===================================================== */
        CASE
            WHEN PrevRentPM IS NULL THEN NULL

            WHEN PrevRentPM = 0 THEN
                'New rent introduced: ' + cast( CAST(CalculatedRentPM AS int) as varchar(40))

            WHEN CalculatedRentPM <> PrevRentPM THEN
                'Rent increased by ' +
                CAST(
                  cast(  ROUND(
                        ((CalculatedRentPM - PrevRentPM) * 100.0) / NULLIF(PrevRentPM,0),
                        0
                    ) as int ) 
                AS VARCHAR(40)) + '%'

            ELSE NULL
        END AS Remarks,ContractStartDate,ContractEndDate,CONVERT(VARCHAR(10), ContractStartDate, 120) 
+ 'To' + 
CONVERT(VARCHAR(10), ContractEndDate, 120) as ContractPeriod, BusinessName, cast(InitialRentPM as int) InitialRentPM ,
			PaymentTermMonths, RiseTermType,  RiseTerm, cast(cast(RiseRate as int) as varchar(10)) + '%' as RiseRate,
			RiseDate, InvoiceDateType,CmdId, BaseId, ClassId
            , 0 as AmountReceived, 0 as AmountReceivable, 0 as Pending, 'Unpaid' as InvoiceStatus

    FROM Segments
)

select * from (
   SELECT 
    COALESCE(e.Id, f.ContractId) AS Id,
    f.ContractId,

    f.ContractNo,
    f.InvoiceNo,

    COALESCE(e.PeriodStart, f.PeriodStart) AS PeriodStart,
    COALESCE(e.PeriodEnd, f.PeriodEnd) AS PeriodEnd,
    COALESCE(e.DueDate, f.DueDate) AS DueDate,

    COALESCE(e.Months, f.Months) AS Months,

    COALESCE(e.CalculatedRentPM, f.CalculatedRentPM) AS CalculatedRentPM,
    COALESCE(e.TotalRent, f.TotalRent) AS TotalRent,

    COALESCE(e.Remarks, f.Remarks) AS Remarks,

    COALESCE(e.ContractStartDate, f.ContractStartDate) AS ContractStartDate,
    COALESCE(e.ContractEndDate, f.ContractEndDate) AS ContractEndDate,
    COALESCE(e.ContractPeriod, f.ContractPeriod) AS ContractPeriod,

    COALESCE(e.BusinessName, f.BusinessName) AS BusinessName,

    COALESCE(e.InitialRentPM, f.InitialRentPM) AS InitialRentPM,
    COALESCE(e.PaymentTermMonths, f.PaymentTermMonths) AS PaymentTermMonths,
    COALESCE(e.RiseTermType, f.RiseTermType) AS RiseTermType,
    COALESCE(e.RiseTerm, f.RiseTerm) AS RiseTerm,
    COALESCE(e.RiseRate, f.RiseRate) AS RiseRate,
    COALESCE(e.RiseDate, f.RiseDate) AS RiseDate,

    COALESCE(e.InvoiceDateType, f.InvoiceDateType) AS InvoiceDateType,

    COALESCE(e.CmdId, f.CmdId) AS CmdId,
    COALESCE(e.BaseId, f.BaseId) AS BaseId,
    COALESCE(e.ClassId, f.ClassId) AS ClassId,

    COALESCE(e.AmountReceived, f.AmountReceived) AS AmountReceived,
    COALESCE(e.AmountReceivable, f.AmountReceivable) AS AmountReceivable,
    COALESCE(e.AmountPending, f.Pending) AS AmountPending,

    COALESCE(e.InvoiceStatus, f.InvoiceStatus) AS InvoiceStatus
	,  ISNULL(e.IsFinalized,0) as IsFinalized
FROM Final f
LEFT JOIN dbo.ContractInvoicesEdit e
    ON e.ContractNo = f.ContractNo
   AND e.InvoiceNo = f.InvoiceNo
   AND ISNULL(e.InvoiceStatus,'') <> 'Deleted'
   and e.SubInvoiceNo is null and e.IsDeleted = 0
   ) cc OUTER APPLY
(
    SELECT rr.TenantNo , rr.RentalValueRate, t.Prefix, t.OwnerName, t.Address, rr.commercialoperationdate as Cod
    FROM dbo.fn_BasicContractInfo(CAST(cc.PeriodStart AS DATE)) rr
	left join Tenants t on t.TenantNo = rr.tenantNo
    WHERE rr.Id = cc.ContractId and t.IsDeleted =0 
) rr
);
GO
