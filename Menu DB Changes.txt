CREATE PROCEDURE dbo.sp_GetContractInvoiceSchedule
    @ContractNo NVARCHAR(50) = NULL,
    @FromDate DATE = NULL,
    @ToDate DATE = NULL
AS
BEGIN
    SELECT *
    FROM dbo.fn_GetContractInvoiceSchedule()
    WHERE (@ContractNo IS NULL OR ContractNo = @ContractNo)
      AND (@FromDate IS NULL OR PeriodStart >= @FromDate)
      AND (@ToDate IS NULL OR PeriodEnd <= @ToDate)
END
